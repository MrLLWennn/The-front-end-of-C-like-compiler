<S> -> <unit>
<unit> -> <external> <unit-tail>
<unit-tail> -> <external> <unit-tail> | empty
<external> -> <record-definition> | <function-or-object>
<record-definition> -> struct identifier { <field-seq> } ;
<function-or-object> -> <base-type> <declarator-stars> identifier <external-tail>
<base-type> -> int | float | double | void | bool | char | struct identifier
<declarator-stars> -> * <declarator-stars> | empty
<external-tail> -> ( <parameters> ) <function-tail> | <array-suffix> <init-opt> ;
<function-tail> -> ; | <block>
<parameters> -> <parameter> <parameter-tail> | empty
<parameter-tail> -> , <parameter> <parameter-tail> | empty
<parameter> -> <base-type> <declarator-stars> identifier <array-suffix>
<array-suffix> -> [ <array-size> ] <array-suffix> | empty
<array-size> -> number | identifier | - number | empty
<field-seq> -> <local-decl> ; <field-seq> | empty
<local-decl> -> <base-type> <declarator-stars> identifier <array-suffix> <init-opt>
<init-opt> -> = <initializer> | empty
<initializer> -> { <initializer> <initializer-tail> } | <expr>
<initializer-tail> -> , <initializer-after-comma> | empty
<initializer-after-comma> -> <initializer> <initializer-tail> | empty
<block> -> { <stmt-seq> }
<stmt-seq> -> <stmt> <stmt-seq> | empty
<stmt> -> <local-decl> ; | <expr-opt> ; | <block> | if ( <expr> ) <stmt> <else-opt> | while ( <expr> ) <block> | for ( <for-init> ; <expr-opt> ; <expr-opt> ) <block> | return <expr-opt> ; | break ; | continue ;
<else-opt> -> else <stmt> | empty
<for-init> -> <local-decl> | <expr-opt>
<expr-opt> -> <expr> | empty
<expr> -> <equality> <assignment-tail>
<assignment-tail> -> <assign-op> <expr> | empty
<assign-op> -> = | += | -= | *= | /=
<equality> -> <relational> <equality-tail>
<equality-tail> -> == <relational> <equality-tail> | != <relational> <equality-tail> | empty
<relational> -> <sum> <relational-tail>
<relational-tail> -> <rel-op> <sum> <relational-tail> | empty
<rel-op> -> < | <= | > | >=
<sum> -> <product> <sum-tail>
<sum-tail> -> + <product> <sum-tail> | - <product> <sum-tail> | empty
<product> -> <unary> <product-tail>
<product-tail> -> * <unary> <product-tail> | / <unary> <product-tail> | empty
<unary> -> + <unary> | - <unary> | & <unary> | * <unary> | ++ <unary> | -- <unary> | <postfix> | ( <base-type> <declarator-stars> ) <unary>
<postfix> -> <atom> <postfix-tail>
<atom> -> identifier | number | character-literal | string-literal | true | false | ( <expr> )
<postfix-tail> -> [ <expr> ] <postfix-tail> | . identifier <postfix-tail> | -> identifier <postfix-tail> | ( <arguments> ) <postfix-tail> | ++ | -- | empty
<arguments> -> <expr> <argument-tail> | empty
<argument-tail> -> , <expr> <argument-tail> | empty
